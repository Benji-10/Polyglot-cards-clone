    47→// ── usePredictiveProgress ──────────────────────────────────
    48→// Drives the progress bar by updating a DOM element directly via a ref,
    49→// bypassing React's state batching which defers updates during async awaits.
    50→// The text % uses React state (updates every ~100ms to avoid jank).
    51→// ──────────────────────────────────────────────────────────
    52→function usePredictiveProgress() {
    53→  const rafRef         = useRef(null)
    54→  const barRef         = useRef(null)   // ref to the <div class="import-progress-fill"> DOM node
    55→  const valueRef       = useRef(0)      // current % value (ground truth)
    56→  const [displayPct, setDisplayPct] = useState(0) // for the text label only
    57→
    58→  const batchTimesRef  = useRef([])
    59→  const batchStartRef  = useRef(null)
    60→  const fromRef        = useRef(0)
    61→  const ceilingRef     = useRef(0)
    62→  const durationRef    = useRef(INITIAL_ESTIMATE_MS)
    63→  const totalRef       = useRef(1)
    64→  const lastTextRef    = useRef(0)      // last time we updated text state
    65→
    66→  const updateBar = useCallback((pct) => {
    67→    valueRef.current = pct
    68→    if (barRef.current) barRef.current.style.width = `${pct}%`
    69→    // Update text label at most every 100ms to avoid React thrashing
    70→    const now = performance.now()
    71→    if (now - lastTextRef.current > 100) {
    72→      lastTextRef.current = now
    73→      setDisplayPct(Math.round(pct))
    74→    }
    75→  }, [])
    76→
    77→  const tick = useCallback(() => {
    78→    if (rafRef.current) cancelAnimationFrame(rafRef.current)
    79→
    80→    const step = () => {
    81→      const elapsed = performance.now() - (batchStartRef.current ?? performance.now())
    82→      const t = Math.min(elapsed / durationRef.current, 1)
    83→      const value = fromRef.current + (ceilingRef.current - fromRef.current) * t
    84→      updateBar(value)
    85→      if (t < 1) rafRef.current = requestAnimationFrame(step)
    86→    }
    87→
    88→    rafRef.current = requestAnimationFrame(step)
    89→  }, [updateBar])
    90→
    91→  const reset = useCallback((totalBatches) => {
    92→    if (rafRef.current) cancelAnimationFrame(rafRef.current)
    93→    totalRef.current       = Math.max(totalBatches, 1)
    94→    valueRef.current       = 0
    95→    batchTimesRef.current  = []
    96→    batchStartRef.current  = null
    97→    fromRef.current        = 0
    98→    ceilingRef.current     = 0
    99→    if (barRef.current) barRef.current.style.width = '0%'
   100→    setDisplayPct(0)
   101→  }, [])
   102→
   103→  const startBatch = useCallback((batchIndex) => {
   104→    batchStartRef.current = performance.now()
   105→    fromRef.current    = (batchIndex / totalRef.current) * 100
   106→    ceilingRef.current = ((batchIndex + 1) / totalRef.current) * 100
   107→    const times = batchTimesRef.current
   108→    durationRef.current = times.length > 0
   109→      ? times.reduce((a, b) => a + b, 0) / times.length
   110→      : INITIAL_ESTIMATE_MS
   111→    tick()
   112→  }, [tick])
   113→
   114→  const completeBatch = useCallback(() => {
   115→    const elapsed = performance.now() - (batchStartRef.current ?? performance.now())
   116→    batchTimesRef.current = [...batchTimesRef.current, elapsed]
   117→    if (rafRef.current) cancelAnimationFrame(rafRef.current)
   118→    updateBar(ceilingRef.current)
   119→    setDisplayPct(Math.round(ceilingRef.current))
   120→  }, [updateBar])
   121→
   122→  useEffect(() => () => { if (rafRef.current) cancelAnimationFrame(rafRef.current) }, [])
   123→  return { barRef, displayPct, reset, startBatch, completeBatch }
   124→}

   185→  const computeFieldChanges = (oldFields, newFields) => {
   186→    const regenFields = []
   187→
   188→    for (const nf of newFields) {
   189→      const of_ = oldFields.find(f => f.key === nf.key)
   190→
   191→      if (!of_) {
   192→        regenFields.push(nf)
   193→        continue
   194→      }
   195→
   196→      const descChanged = of_.description !== nf.description
   197→      const typeChanged = of_.field_type  !== nf.field_type
   198→
   199→      // Phonetics: regen whenever the annotation set changes in a way that needs new data.
   200→      // - Ruby changed to any different value (including switching between two non-'none' values)
   201→      // - Any extra added (was not there before)
   202→      // Does NOT regen when annotations are only removed (existing data stays in DB, just not shown).
   203→      const oldPh = normalisePhonetics(of_.phonetics)
   204→      const newPh = normalisePhonetics(nf.phonetics)
   205→      const rubyChanged = newPh.ruby !== oldPh.ruby && newPh.ruby !== 'none'
   206→      const extrasAdded = newPh.extras.some(k => !oldPh.extras.includes(k))
   207→      const phonNeedsRegen = rubyChanged || extrasAdded
   208→
   209→      if (descChanged || typeChanged || phonNeedsRegen) {
   210→        regenFields.push({
   211→          ...nf,
   212→          _forceRegen: descChanged || typeChanged,
   213→        })
   214→      }
   215→    }
   216→
   217→    // Keys removed from blueprint → purge their data from all cards
   218→    const newKeys = new Set(newFields.map(f => f.key))
   219→    const deletedKeys = oldFields
   220→      .filter(f => !MANDATORY_FIELD_KEYS.includes(f.key) && !newKeys.has(f.key))
   221→      .map(f => f.key)
   222→
   223→    // Also compute annotation sub-keys that were removed (e.g., fieldKey_english)
   224→    // so we can delete them from card data
   225→    const deletedAnnotationKeys = []
   226→    for (const nf of newFields) {
   227→      const of_ = oldFields.find(f => f.key === nf.key)
   228→      if (!of_) continue
   229→      const oldPh = normalisePhonetics(of_.phonetics)
   230→      const newPh = normalisePhonetics(nf.phonetics)
   231→      if (oldPh.ruby !== 'none' && oldPh.ruby !== newPh.ruby) {
   232→        deletedAnnotationKeys.push(`${nf.key}_${oldPh.ruby}`)
   233→      }
   234→      oldPh.extras.forEach(k => {
   235→        if (!newPh.extras.includes(k)) deletedAnnotationKeys.push(`${nf.key}_${k}`)
   236→      })
   237→    }
   238→
   239→    return { regenFields, deletedKeys, deletedAnnotationKeys }
   240→  }
   241→
   242→  const runBackgroundRegen = async (savedFields, _allCards) => {
   243→    const oldFields = originalBlueprintRef.current || []
   244→    const { regenFields, deletedKeys, deletedAnnotationKeys } = computeFieldChanges(oldFields, savedFields)
   245→
   246→    // Delete removed field data and removed annotation sub-keys from cards
   247→    const allDeleted = [...deletedKeys, ...deletedAnnotationKeys]
   248→    if (allDeleted.length > 0) {
   249→      api.deleteCardFields(deckId, allDeleted).catch(e => console.error('Field delete error:', e))
   250→    }
   251→
   252→    if (regenFields.length === 0) {
   253→      if (allDeleted.length > 0) qc.invalidateQueries({ queryKey: ['cards', deckId] })
   254→      originalBlueprintRef.current = savedFields
   255→      return
   256→    }
   257→
   258→    // Fetch fresh cards — avoids stale cache
   259→    let freshCards
   260→    try { freshCards = await api.getCards(deckId) }
   261→    catch (e) { toast.error(`Could not fetch cards for refresh: ${e.message}`); return }
   262→    if (!freshCards.length) { originalBlueprintRef.current = savedFields; return }
   263→
   264→    // Only regen cards that are actually missing at least one regen field
   265→    const getAnnotationKeys = (ph) => {
   266→      if (!ph) return []
   267→      if (Array.isArray(ph)) return ph.filter(k => k && k !== 'none')
   268→      const keys = []
   269→      if (ph.ruby && ph.ruby !== 'none') keys.push(ph.ruby)
   270→      if (Array.isArray(ph.extras)) keys.push(...ph.extras)
   271→      return keys
   272→    }
   273→    const fieldNeedsRegen = (card, field) => {
   274→      if (field._forceRegen) return true
   275→      const v = card.fields?.[field.key]
   276→      if (!v || v === '') return true
   277→
   278→      const annotationKeys = getAnnotationKeys(field.phonetics)
   279→      if (annotationKeys.length === 0) {
   280→        if (field.field_type === 'example') {
   281→          const text = typeof v === 'object' ? v.text : v
   282→          return !text
   283→        }
   284→        return false
   285→      }
   286→      if (typeof v !== 'object') return true
   287→      if (!v.text) return true
   288→      return annotationKeys.some(ak => !v[ak])
   289→    }
   290→    const cardsNeedingRegen = freshCards.filter(card =>
   291→      regenFields.some(f => fieldNeedsRegen(card, f))
   292→    )
   293→    if (cardsNeedingRegen.length === 0) {
   294→      if (deletedKeys.length > 0) qc.invalidateQueries({ queryKey: ['cards', deckId] })
   295→      originalBlueprintRef.current = savedFields
   296→      return
   297→    }
   298→
   299→    // Use the same batch size as AI import — keeps regen and import batch counts comparable
   300→    const batches = []
   301→    for (let i = 0; i < cardsNeedingRegen.length; i += BATCH_SIZE) {
   302→      batches.push(cardsNeedingRegen.slice(i, i + BATCH_SIZE))
   303→    }
   304→
   305→    setRegenState('running')
   306→    setRegenMsg(`Refreshing ${regenFields.length} field${regenFields.length !== 1 ? 's' : ''} for ${cardsNeedingRegen.length} card${cardsNeedingRegen.length !== 1 ? 's' : ''} (${batches.length} batch${batches.length !== 1 ? 'es' : ''})…`)
   307→    updateRegenBar(0)
   308→
   309→    let totalPatched = 0
   310→    let completedBatches = 0
   311→    const failedBatchNums = []
   312→    const targetLang = deckRef.current?.target_language || deck?.target_language || ''
   313→    const allPatches = []
   314→
   315→    // Helper: process one batch with independent retry
   316→    const processBatch = async (batchIdx, batchCards) => {
   317→      const vocabItems = batchCards.map(card => {
   318→        const sense = card.fields?.context || card.fields?.source_translation || ''
   319→        return sense ? `${card.word} (${sense})` : card.word
   320→      })
   321→
   322→      const MAX_RETRIES = 4
   323→      let genRes = null
   324→      let lastErr = null
   325→      for (let attempt = 0; attempt <= MAX_RETRIES; attempt++) {
   326→        if (attempt > 0) {
   327→          const delay = 1500 * Math.pow(2, attempt - 1)
   328→          await new Promise(r => setTimeout(r, delay))
   329→        }
   330→        try {
   331→          genRes = await api.generateCards(deckId, {
   332→            vocab: vocabItems,
   333→            targetLanguage: targetLang,
   334→            sourceLanguage: deckRef.current?.source_language || 'English',
   335→            contextLanguage: deckRef.current?.context_language || 'target',
   336→          }, regenFields)
   337→          if (genRes?.cards?.length) { lastErr = null; break }
   338→          lastErr = new Error('No cards returned')
   339→        } catch (e) { lastErr = e }
   340→      }
   341→
   342→      if (lastErr || !genRes?.cards?.length) {
   343→        console.error(`Regen batch ${batchIdx + 1} failed:`, lastErr?.message)
   344→        failedBatchNums.push(batchIdx + 1)
   345→        return
   346→      }
   347→
   348→      // With the nested annotation shape, Gemini returns each field as either:
   349→      //   - a plain string (source_translation, context, unannotated fields)
   350→      //   - an object { text, annotationType, ... } (annotated/example fields)
   351→      // We patch by field key — the deep-merge in cards-patch handles nested merging.
   352→      const changedFieldKeys = regenFields.map(f => f.key)
   353→
   354→      genRes.cards.filter(c => !c._error).forEach((genCard, idx) => {
   355→        const card = batchCards[idx]
   356→        if (!card) return
   357→        const fieldData = {}
   358→        changedFieldKeys.forEach(k => {
   359→          const v = genCard[k]
   360→          if (v !== undefined && v !== null && v !== '') fieldData[k] = v
   361→        })
   362→        if (regenFields.some(f => fieldData[f.key] != null)) {
   363→          allPatches.push({ id: card.id, word: card.word, fields: fieldData })
   364→        }
   365→      })
   366→    }
   367→
   368→    // Run batches 2 at a time (parallel) — doubles throughput within rate limit
   369→    const CONCURRENCY = 2
   370→    for (let i = 0; i < batches.length; i += CONCURRENCY) {
   371→      const chunk = batches.slice(i, i + CONCURRENCY)
   372→      const startBatchIdx = i
   373→
   374→      // Animate progress bar across this chunk
   375→      const chunkStart = (i / batches.length) * 90 // reserve last 10% for saving
   376→      const chunkEnd   = (Math.min(i + CONCURRENCY, batches.length) / batches.length) * 90
   377→      const startTime  = performance.now()
   378→      const ESTIMATE   = 12000 // slightly longer — two batches in parallel
   379→
   380→      if (regenRafRef.current) cancelAnimationFrame(regenRafRef.current)
   381→      const tickRegen = () => {
   382→        const t = Math.min((performance.now() - startTime) / ESTIMATE, 0.9)
   383→        updateRegenBar(chunkStart + (chunkEnd - chunkStart) * t)
   384→        if (t < 0.9) regenRafRef.current = requestAnimationFrame(tickRegen)
   385→      }
   386→      regenRafRef.current = requestAnimationFrame(tickRegen)
   387→      setRegenMsg(`Batches ${startBatchIdx + 1}–${Math.min(startBatchIdx + CONCURRENCY, batches.length)} / ${batches.length} — generating…`)
   388→
   389→      await Promise.all(chunk.map((batchCards, j) => processBatch(startBatchIdx + j, batchCards)))
   390→
   391→      if (regenRafRef.current) cancelAnimationFrame(regenRafRef.current)
   392→      completedBatches += chunk.length
   393→      updateRegenBar((completedBatches / batches.length) * 90)
   394→    }
   395→
   396→    // Saving phase — the last 10% of the bar
   397→    if (allPatches.length > 0) {
   398→      setRegenMsg(`Saving ${allPatches.length} card${allPatches.length !== 1 ? 's' : ''}…`)
   399→      updateRegenBar(90)
   400→      try {
   401→        await api.patchCards(deckId, allPatches)
   402→        totalPatched = allPatches.length
   403→      } catch (e) {
   404→        console.error('Regen patch failed:', e)
   405→        failedBatchNums.push('save')
   406→      }
   407→    }
   408→
   409→    qc.invalidateQueries({ queryKey: ['cards', deckId] })
   410→    updateRegenBar(100)
   411→
   412→    const anyFailed = failedBatchNums.length > 0
   413→    if (anyFailed) {
   414→      originalBlueprintRef.current = oldFields
   415→      setRegenState('error')
   416→      setRegenMsg(`${totalPatched} cards updated. Failures: batch${failedBatchNums.length > 1 ? 'es' : ''} ${failedBatchNums.join(', ')} — save again to retry.`)
   417→      toast.error(`Some card updates failed. Save the blueprint again to retry.`)
   418→    } else {
   419→      originalBlueprintRef.current = savedFields
   420→      setRegenState('done')
   421→      setRegenMsg(`${totalPatched} card${totalPatched !== 1 ? 's' : ''} refreshed`)
   422→    }
   423→  }
   424→
   425→  const allCardsRef = useRef([])
   426→  const { data: allCards = [] } = useQuery({
   427→    queryKey: ['cards', deckId],
   428→    queryFn: () => api.getCards(deckId),
   429→    enabled: !!deckId,

   437→      setFields(normalised)
   438→      qc.setQueryData(['blueprint', deckId], saved)
   439→      toast.success('Blueprint saved!')
   440→      // Kick off background regeneration if there are cards to update
   441→      if (allCardsRef.current.length > 0) {
   442→        runBackgroundRegen(normalised, allCardsRef.current)
   443→      }
   444→    },
   445→    onError: e => toast.error(`Save failed: ${e.message}`),
   446→  })
   447→
   448→  const batchSaveMutation = useMutation({
   449→    mutationFn: cards => api.batchCreateCards(cards),
   450→    onSuccess: () => qc.invalidateQueries({ queryKey: ['cards', deckId] }),
   451→    onError: e => toast.error(`Save error: ${e.message}`),
   452→  })

   454→  // ── Field management ────────────────────────────────────
   455→  // Keys are stable identifiers generated once at creation — they never change,
   456→  // even when the user renames the label.
   457→  const makeKey = () => `f_${Math.random().toString(36).slice(2, 8)}`
   458→
   459→  const addField = suggested => {
   460→    const base = suggested
   461→      ? { ...suggested, key: suggested.key || makeKey(), phonetics: normalisePhonetics(suggested.phonetics) }
   462→      : { key: makeKey(), label: 'New Field', description: '', field_type: 'text', show_on_front: false, phonetics: { ruby: 'none', extras: [] } }
   463→    setFields(prev => [...prev, { ...base, position: prev.length }])
   464→  }
   465→
   466→  const updateField = (idx, patch) => setFields(prev => prev.map((f, i) => i === idx ? { ...f, ...patch } : f))
   467→  const removeField = idx => setFields(prev => prev.filter((_, i) => i !== idx))
   468→  const moveField = (idx, dir) => {
   469→    setFields(prev => {
   470→      const next = [...prev]
   471→      const swap = idx + dir
   472→      if (swap < 0 || swap >= next.length) return prev
   473→      ;[next[idx], next[swap]] = [next[swap], next[idx]]
   474→      return next
   475→    })
   476→  }

   481→  useEffect(() => { deckRef.current = deck }, [deck])

   483→  // ── CSV Import (AI) ──────────────────────────────────────
   484→  const handleCSV = useCallback(file => {
   485→    setImportState('generating')
   486→    setImportError(null)
   487→    setTotalImported(0)
   488→
   489→    Papa.parse(file, {
   490→      complete: async res => {
   491→        try {
   492→          const vocab = res.data.flat().map(v => String(v).trim()).filter(Boolean)
   493→          if (!vocab.length) { setImportError('No words found in CSV'); setImportState('error'); return }
   494→
   495→          // Warn if the CSV looks like a structured file (multiple populated columns per row)
   496→          // — the AI importer expects a flat word list, not headers + data
   497→          const rows = res.data.filter(r => Array.isArray(r) && r.some(c => String(c).trim()))
   498→          if (rows.length >= 2) {
   499→            const firstRowCols = rows[0].filter(c => String(c).trim()).length
   500→            const secondRowCols = rows[1].filter(c => String(c).trim()).length
   501→            if (firstRowCols > 1 && secondRowCols > 1) {
   502→              setImportError(
   503→                'This looks like a structured CSV with multiple columns. ' +
   504→                'The AI importer expects a plain word list (one word per cell). ' +
   505→                'Use the "Direct CSV" tab if you want to import a structured file.'
   506→              )
   507→              setImportState('error')
   508→              return
   509→            }
   510→          }
   511→
   512→          const batches = []
   513→          for (let i = 0; i < vocab.length; i += BATCH_SIZE) batches.push(vocab.slice(i, i + BATCH_SIZE))
   514→          progress.reset(batches.length)
   515→          setProgressMsg('Generating cards…')
   516→
   517→          let allGeneratedCards = []
   518→          let failedBatchCount = 0
   519→
   520→          // Helper: fetch one batch with independent exponential backoff retry
   521→          const fetchBatch = async (batchIdx) => {
   522→            const MAX_RETRIES = 4
   523→            let genRes = null
   524→            let lastErr = null
   525→            for (let attempt = 0; attempt <= MAX_RETRIES; attempt++) {
   526→              if (attempt > 0) {
   527→                const delay = 1500 * Math.pow(2, attempt - 1)
   528→                await new Promise(r => setTimeout(r, delay))
   529→              }
   530→              try {
   531→                genRes = await api.generateCards(deckId, {
   532→                  vocab: batches[batchIdx],
   533→                  targetLanguage: deckRef.current?.target_language || deck?.target_language || '',
   534→                  sourceLanguage: deckRef.current?.source_language || 'English',
   535→                  contextLanguage: deckRef.current?.context_language || 'target',
   536→                }, fieldsRef.current || [])
   537→                if (genRes?.cards?.length) { lastErr = null; break }
   538→                lastErr = new Error('No cards returned from API')
   539→              } catch (e) { lastErr = e }
   540→            }
   541→            return { batchIdx, genRes, lastErr }
   542→          }
   543→
   544→          // Validate cards from a batch result
   545→          const requiredKeys = (fieldsRef.current || []).map(f => f.key).filter(k => !MANDATORY_FIELD_KEYS.includes(k))
   546→          const validateCards = (cards) => cards.filter(c => {
   547→            if (c._error || !c.word) return false
   548→            if (requiredKeys.length === 0) return true
   549→            return requiredKeys.some(k => c[k] && c[k] !== '')
   550→          })
   551→
   552→          // Rolling pipeline: 2 batches in-flight at once.
   553→          // As soon as any slot frees up, the next batch starts immediately.
   554→          const CONCURRENCY = 2
   555→          let nextBatch = 0
   556→          let completedCount = 0
   557→
   558→          await new Promise(resolve => {
   559→            const launch = () => {
   560→              while (nextBatch < batches.length && (nextBatch - completedCount) < CONCURRENCY) {
   561→                const idx = nextBatch++
   562→                progress.startBatch(idx)
   563→                fetchBatch(idx).then(({ genRes, lastErr }) => {
   564→                  completedCount++
   565→                  if (lastErr || !genRes?.cards?.length) {
   566→                    failedBatchCount++
   567→                  } else {
   568→                    allGeneratedCards.push(...validateCards(genRes.cards))
   569→                    setProgressMsg(`${allGeneratedCards.length} card${allGeneratedCards.length !== 1 ? 's' : ''} generated…`)
   570→                  }
   571→                  progress.completeBatch(idx)
   572→                  if (completedCount >= batches.length) resolve()
   573→                  else launch()
   574→                })
   575→              }
   576→            }
   577→            launch()
   578→            if (batches.length === 0) resolve()
   579→          })
   580→
   581→          if (!allGeneratedCards.length) {
   582→            setImportError(failedBatchCount > 0 ? `All ${failedBatchCount} batch${failedBatchCount !== 1 ? 'es' : ''} failed — check your connection and try again` : 'No valid cards generated')
   583→            setImportState('error')
   584→            return
   585→          }
   586→
   587→          // ── Saving phase — update progress bar for this step ──
   588→          setProgressMsg(`Saving ${allGeneratedCards.length} cards…`)
   589→
   590→          // ── Detect homographs ──────────────────────────────
   591→          const wordGroups = {}
   592→          for (const card of allGeneratedCards) {
   593→            if (!wordGroups[card.word]) wordGroups[card.word] = []
   594→            wordGroups[card.word].push(card)
   595→          }
   596→
   597→          const homographGroups = Object.entries(wordGroups)
   598→            .filter(([, cards]) => cards.length > 1)
   599→            .map(([word, cards]) => ({
   600→              word,
   601→              senses: cards.map(c => ({ card: c, selected: true, sense: c._sense || '' })),
   602→            }))
   603→
   604→          // Function that detects collisions then saves
   605→          const proceedToCollisionCheck = async (cardsToSave) => {
   606→            // Build word → array of existing cards (a word can have multiple for homographs)
   607→            const existingByWord = {}
   608→            for (const c of allCardsRef.current) {
   609→              if (!existingByWord[c.word]) existingByWord[c.word] = []
   610→              existingByWord[c.word].push(c)
   611→            }
   612→
   613→            // For each incoming card, find the best matching existing card.
   614→            // Match by _sense if available (context/source_translation field), then by index.
   615→            const usedExistingIds = new Set()
   616→            const collisions = []
   617→            const noCollision = []
   618→
   619→            for (const incoming of cardsToSave) {
   620→              const existingArr = existingByWord[incoming.word]
   621→              if (!existingArr || existingArr.length === 0) {
   622→                noCollision.push(incoming)
   623→                continue
   624→              }
   625→
   626→              // Try to match by sense (context or source_translation field)
   627→              const incomingSense = (incoming._sense || incoming.source_translation || '').toLowerCase().trim()
   628→              let match = null
   629→
   630→              if (incomingSense) {
   631→                match = existingArr.find(e => {
   632→                  if (usedExistingIds.has(e.id)) return false
   633→                  const eSense = (e.fields?.context || e.fields?.source_translation || '').toLowerCase().trim()
   634→                  return eSense === incomingSense || eSense.includes(incomingSense) || incomingSense.includes(eSense)
   635→                })
   636→              }
   637→
   638→              // Fall back to first unmatched existing card
   639→              if (!match) match = existingArr.find(e => !usedExistingIds.has(e.id))
   640→
   641→              if (match) {
   642→                usedExistingIds.add(match.id)
   643→                collisions.push({ incoming, existing: match })
   644→              } else {
   645→                // All existing cards for this word are already matched — this is a new sense
   646→                noCollision.push(incoming)
   647→              }
   648→            }
   649→
   650→            if (collisions.length === 0) {
   651→              await saveCards(noCollision)
   652→            } else {
   653→              setImportState('review')
   654→              setCollisionPending({
   655→                collisions,
   656→                noCollision,
   657→                blueprint: fieldsRef.current || [],
   658→                onConfirm: async (decisions) => {
   659→                  setCollisionPending(null)
   660→                  setImportState('generating')
   661→
   662→                  const toAdd = [...noCollision]
   663→                  const toUpdate = []
   664→                  for (const { incoming, existing, action } of decisions) {
   665→                    if (action === 'add')    toAdd.push(incoming)
   666→                    if (action === 'update') toUpdate.push({ incoming, existing })
   667→                  }
   668→
   669→                  if (toUpdate.length) {
   670→                    // Patch by id — correct for homographs
   671→                    const patches = toUpdate.map(({ incoming, existing }) => {
   672→                      const { _meanings, _sense, _error, word, ...fields } = incoming
   673→                      return { id: existing.id, word, fields }
   674→                    })
   675→                    await api.patchCards(deckId, patches).catch(() => {})
   676→                  }
   677→
   678→                  await saveCards(toAdd)
   679→                },
   680→              })
   681→            }
   682→          }
   683→
   684→          // Function that actually saves the chosen cards
   685→          const saveCards = async (cardsToSave) => {
   686→            const toSave = cardsToSave.map(c => {
   687→              // Strip internal meta fields — everything else is card field data
   688→              const { _meanings, _sense, _error, word, ...fields } = c
   689→              return { deck_id: deckId, word, fields }
   690→            })
   691→            if (toSave.length) {
   692→              await batchSaveMutation.mutateAsync(toSave)
   693→              setTotalImported(prev => prev + toSave.length)
   694→            }
   695→            setImportState('done')
   696→            qc.invalidateQueries({ queryKey: ['cards', deckId] })
   697→          }
   698→
   699→          if (homographGroups.length === 0) {
   700→            // No homographs — check collisions then save
   701→            await proceedToCollisionCheck(allGeneratedCards)
   702→          } else {
   703→            // Pause import and show homograph review modal
   704→            setImportState('review')
   705→            setHomographPending({
   706→              groups: homographGroups,
   707→              allCards: allGeneratedCards,
   708→              onConfirm: async (approvedHomographCards) => {
   709→                setHomographPending(null)
   710→                setImportState('generating')
   711→                // Replace homograph words with only the approved senses
   712→                const homographWords = new Set(homographGroups.map(g => g.word))
   713→                const nonHomographCards = allGeneratedCards.filter(c => !homographWords.has(c.word))
   714→                await proceedToCollisionCheck([...nonHomographCards, ...approvedHomographCards])
   715→              },
   716→            })
   717→          }
   718→        } catch (e) { setImportError(e.message); setImportState('error') }
   719→      },
   720→      error: e => { setImportError(e.message); setImportState('error') },
   721→    })
   722→  }, [deckId, progress]) // eslint-disable-line
   723→
   724→  const handleDrop = useCallback(e => { e.preventDefault(); const f = e.dataTransfer.files[0]; if (f) handleCSV(f) }, [handleCSV])
   725→
   726→  // ── Direct CSV import (no AI) ───────────────────────────
   727→  // Row 1: headers — word, [field keys], optional SRS columns
   728→  // Row 2: metadata JSON per field column (label, description, field_type, show_on_front, phonetics)
   729→  //        — only present in files exported from this app
   730→  // Row 3+: card data
   731→  const handleDirectCSV = useCallback(file => {
   732→    setImportState('generating')
   733→    setImportError(null)
   734→    setTotalImported(0)
   735→
   736→    Papa.parse(file, {
   737→      header: false,       // parse raw rows so we can handle the metadata row ourselves
   738→      skipEmptyLines: true,
   739→      complete: async res => {
   740→        try {
   741→          const allRows = res.data
   742→          if (allRows.length < 2) { setImportError('CSV needs at least a header and one data row'); setImportState('error'); return }
   743→
   744→          const headers = allRows[0].map(h => String(h).trim())
   745→          if (!headers.includes('word')) { setImportError('CSV must have a "word" column'); setImportState('error'); return }
   746→
   747→          const SRS_COLS = new Set(['srs_state', 'last_reviewed', 'interval', 'stability', 'difficulty', 'repetitions', 'seen'])
   748→
   749→          // Detect metadata row: row 2 where at least one blueprint field column contains JSON with "label"
   750→          let metaRow = null
   751→          let dataStartIdx = 1
   752→          const row2 = allRows[1]
   753→          const hasMeta = row2.some((cell, i) => {
   754→            if (!cell || headers[i] === 'word' || SRS_COLS.has(headers[i])) return false
   755→            try { const p = JSON.parse(cell); return typeof p === 'object' && 'label' in p } catch { return false }
   756→          })
   757→          if (hasMeta) { metaRow = row2; dataStartIdx = 2 }
   758→
   759→          const fieldCols = headers.filter(h => h !== 'word' && !SRS_COLS.has(h))
   760→
   761→          // If we have a metadata row, update the blueprint with the imported field definitions
   762→          if (metaRow) {
   763→            const newFields = []
   764→            for (let i = 0; i < headers.length; i++) {
   765→              const h = headers[i]
   766→              if (h === 'word' || SRS_COLS.has(h) || MANDATORY_FIELD_KEYS.includes(h)) continue
   767→              try {
   768→                const meta = JSON.parse(metaRow[i] || '{}')
   769→                newFields.push({
   770→                  key:           h,
   771→                  label:         meta.label        || h,
   772→                  description:   meta.description  || '',
   773→                  field_type:    meta.field_type   || 'text',
   774→                  show_on_front: meta.show_on_front || false,
   775→                  phonetics:     normalisePhonetics(meta.phonetics),
   776→                  position:      newFields.length,
   777→                })
   778→              } catch {
   779→                newFields.push({ key: h, label: h, description: '', field_type: 'text', show_on_front: false, phonetics: { ruby: 'none', extras: [] }, position: newFields.length })
   780→              }
   781→            }
   782→            if (newFields.length > 0) {
   783→              // Keep mandatory fields, prepend them, save blueprint
   784→              const mandatory = (fieldsRef.current || []).filter(f => MANDATORY_FIELD_KEYS.includes(f.key))
   785→              const merged = [...mandatory, ...newFields]
   786→              try {
   787→                await api.saveBlueprintFields(deckId, merged)
   788→                qc.invalidateQueries({ queryKey: ['blueprint', deckId] })
   789→                setFields(merged.map(f => ({ ...f, phonetics: normalisePhonetics(f.phonetics) })))
   790→                originalBlueprintRef.current = merged
   791→              } catch (e) {
   792→                console.error('Blueprint update failed:', e)
   793→              }
   794→            }
   795→          }
   796→
   797→          // Parse data rows into cards
   798→          const wordIdx = headers.indexOf('word')
   799→          const colIdx  = (col) => headers.indexOf(col)
   800→          const cardsToSave = allRows.slice(dataStartIdx)
   801→            .filter(row => row[wordIdx]?.trim())
   802→            .map(row => {
   803→              const card = { deck_id: deckId, word: row[wordIdx].trim(), fields: {} }
   804→              for (const col of fieldCols) {
   805→                const idx = colIdx(col)
   806→                if (idx >= 0 && row[idx] !== undefined && row[idx] !== '') {
   807→                  const raw = row[idx]
   808→                  // Field values that are objects were exported as JSON — try to parse them back
   809→                  try {
   810→                    const parsed = JSON.parse(raw)
   811→                    card.fields[col] = (parsed && typeof parsed === 'object') ? parsed : raw
   812→                  } catch {
   813→                    card.fields[col] = raw
   814→                  }
   815→                }
   816→              }
   817→              if (row[colIdx('srs_state')])     card.srs_state     = row[colIdx('srs_state')]
   818→              if (row[colIdx('last_reviewed')]) card.last_reviewed  = row[colIdx('last_reviewed')]
   819→              if (row[colIdx('interval')])      card.interval       = Number(row[colIdx('interval')]) || 0
   820→              if (row[colIdx('stability')])     card.stability      = Number(row[colIdx('stability')]) || 0
   821→              if (row[colIdx('difficulty')])    card.difficulty     = Number(row[colIdx('difficulty')]) || 5
   822→              if (row[colIdx('repetitions')])   card.repetitions    = Number(row[colIdx('repetitions')]) || 0
   823→              return card
   824→            })
   825→
   826→          if (!cardsToSave.length) { setImportError('No valid data rows found'); setImportState('error'); return }
   827→
   828→          // Collision check — word → array to handle homographs
   829→          const existingByWord = {}
   830→          for (const c of allCardsRef.current) {
   831→            if (!existingByWord[c.word]) existingByWord[c.word] = []
   832→            existingByWord[c.word].push(c)
   833→          }
   834→          const usedIds = new Set()
   835→          const collisions = []
   836→          const noCollision = []
   837→          for (const incoming of cardsToSave) {
   838→            const arr = existingByWord[incoming.word]
   839→            if (!arr?.length) { noCollision.push(incoming); continue }
   840→            const match = arr.find(e => !usedIds.has(e.id))
   841→            if (match) { usedIds.add(match.id); collisions.push({ incoming, existing: match }) }
   842→            else noCollision.push(incoming)
   843→          }
   844→
   845→          const saveDirect = async (toAdd, toUpdate) => {
   846→            if (toUpdate.length) {
   847→              const patches = toUpdate.map(({ incoming, existing }) => ({
   848→                id: existing.id, word: incoming.word, fields: incoming.fields,
   849→              }))
   850→              await api.patchCards(deckId, patches).catch(() => {})
   851→            }
   852→            if (toAdd.length) {
   853→              await batchSaveMutation.mutateAsync(toAdd)
   854→              setTotalImported(toAdd.length)
   855→            }
   856→            qc.invalidateQueries({ queryKey: ['cards', deckId] })
   857→            setImportState('done')
   858→          }
   859→
   860→          if (collisions.length === 0) {
   861→            await saveDirect(cardsToSave, [])
   862→          } else {
   863→            setImportState('review')
   864→            setCollisionPending({
   865→              collisions,
   866→              noCollision,
   867→              blueprint: fieldsRef.current || [],
   868→              onConfirm: async (decisions) => {
   869→                setCollisionPending(null)
   870→                setImportState('generating')
   871→                const toAdd = [...noCollision]
   872→                const toUpdate = []
   873→                for (const { incoming, existing, action } of decisions) {
   874→                  if (action === 'add')    toAdd.push(incoming)
   875→                  if (action === 'update') toUpdate.push({ incoming, existing })
   876→                }
   877→                await saveDirect(toAdd, toUpdate)
   878→              },
   879→            })
   880→          }
   881→        } catch (e) { setImportError(e.message); setImportState('error') }
   882→      },
   883→      error: e => { setImportError(e.message); setImportState('error') },
   884→    })
   885→  }, [deckId, batchSaveMutation, qc]) // eslint-disable-line
   886→
   887→  if (blueprintLoading || fields === null) {
   888→    return (
   889→      <div className="max-w-3xl mx-auto px-6 py-10">
   890→        <div className="section-title mb-1">Blueprint & Import</div>
   891→        <div className="font-display text-3xl font-bold mb-8" style={{ color: 'var(--text-primary)' }}>{deck?.name || '...'}</div>
   892→        <div className="space-y-3">{[1,2,3].map(i => <div key={i} className="h-14 rounded-xl shimmer" />)}</div>
   893→      </div>
   894→    )
   895→  }
   896→
   897→  const usedKeys = new Set(fields.map(f => f.key))
   898→  const quickAdd = (settings.quickAddFields || []).filter(s => !usedKeys.has(s.key))
   899→
   900→  return (
   901→    <div className="max-w-3xl mx-auto px-6 py-10">
   902→      <div className="mb-8">
   903→        <div className="section-title mb-1">Blueprint & Import</div>
   904→        <h1 className="font-display text-3xl font-bold mb-2" style={{ color: 'var(--text-primary)' }}>{deck?.name || 'Deck'}</h1>
   905→        <p className="text-sm" style={{ color: 'var(--text-muted)' }}>Define fields Gemini fills for each card, then import your vocab.</p>
   906→      </div>
   907→
   908→      {/* ── Fields ─────────────────────────────────────── */}
   909→      <section className="mb-10">
   910→        <div className="flex items-center justify-between mb-4">
   911→          <h2 className="font-display font-semibold text-lg" style={{ color: 'var(--text-primary)' }}>Card Fields</h2>
   912→          <button className="btn-primary text-xs py-1.5 px-3" onClick={() => addField()}>+ Add Field</button>
   913→        </div>
   914→
   915→        {fields.length === 0 ? (
   916→          <div className="text-center py-10 card rounded-2xl mb-4">
   917→            <div className="text-3xl mb-3">🗺</div>
   918→            <div className="font-medium mb-1" style={{ color: 'var(--text-secondary)' }}>No fields yet</div>
   919→            <div className="text-xs" style={{ color: 'var(--text-muted)' }}>Add fields or use quick-add below</div>
   920→          </div>
   921→        ) : (
   922→          <div className="space-y-2 mb-4">
   923→            {fields.map((field, idx) => (
   924→              <FieldRow key={idx} field={field}
   925→                onUpdate={p => updateField(idx, p)}
   926→                onRemove={() => removeField(idx)}
   927→                onMoveUp={() => moveField(idx, -1)}
   928→                onMoveDown={() => moveField(idx, 1)}
   929→                isFirst={idx === 0} isLast={idx === fields.length - 1}
   930→              />
   931→            ))}
   932→          </div>
   933→        )}
   934→
   935→        {quickAdd.length > 0 && (
   936→          <div className="mb-5">
   937→            <div className="section-title mb-2">Quick add</div>
   938→            <div className="flex flex-wrap gap-2">
   939→              {quickAdd.map(s => (
   940→                <button key={s.key} className="tag cursor-pointer transition-all hover:border-purple-500" onClick={() => addField(s)}>
   941→                  + {s.label}
   942→                </button>
   943→              ))}
   944→            </div>
   945→          </div>
   946→        )}
   947→
   948→        <div className="flex items-center gap-3">
   949→          <button className="btn-primary" disabled={saveMutation.isPending} onClick={() => saveMutation.mutate(fields)}>
   950→            {saveMutation.isPending ? 'Saving...' : '✓ Save Blueprint'}
   951→          </button>
   952→          {saveMutation.isSuccess && regenState === 'idle' && (
   953→            <span className="text-sm" style={{ color: 'var(--accent-secondary)' }}>Saved!</span>
   954→          )}
   955→        </div>
   956→
   957→        {/* Background regen progress */}
   958→        {regenState !== 'idle' && (
   959→          <div className="mt-3 rounded-xl p-4" style={{ background: 'var(--bg-surface)', border: '1px solid var(--border)' }}>
   960→            <div className="flex items-center gap-3 mb-2">
   961→              <span className="text-sm" style={{ animation: regenState === 'running' ? 'pulse 1.5s ease-in-out infinite' : 'none' }}>
   962→                {regenState === 'done' ? '✅' : regenState === 'error' ? '❌' : '⚡'}
   963→              </span>
   964→              <div className="flex-1 min-w-0">
   965→                <div className="text-xs font-medium" style={{ color: 'var(--text-primary)' }}>
   966→                  {regenState === 'done' ? 'Cards updated' : 'Updating existing cards…'}
   967→                </div>

  1186→function ensureMandatoryFields(fields) {
  1187→  const result = [...fields]
  1188→  for (const key of [...MANDATORY_FIELD_KEYS].reverse()) {

  1190→      result.unshift({ ...MANDATORY_DEFAULTS[key], phonetics: normalisePhonetics(MANDATORY_DEFAULTS[key].phonetics) })

  1197→function normalisePhonetics(ph) {
  1198→  if (!ph) return { ruby: 'none', extras: [] }
  1199→  if (Array.isArray(ph)) {
  1200→    const RUBY_KEYS = RUBY_OPTIONS.map(o => o.key).filter(k => k !== 'none')
  1201→    const ruby = ph.find(k => RUBY_KEYS.includes(k)) || 'none'
  1202→    const extras = ph.filter(k => !RUBY_KEYS.includes(k))
  1203→    return { ruby, extras }
  1204→  }
  1205→  if (typeof ph === 'object') return { ruby: ph.ruby || 'none', extras: ph.extras || [] }
  1206→  return { ruby: 'none', extras: [] }

  1372→    try {
  1373→      // Wrap annotated/example field values as [{ text, annotations }]
  1374→      const builtFields = {}
  1375→      for (const f of fields) {
  1376→        const text = fieldValues[f.key] || ''
  1377→        if (!text) continue
  1378→        const annotationKeys = getAnnotationKeys(f.phonetics)
  1379→        const isStructured = annotationKeys.length > 0 || f.field_type === 'example'
  1380→        if (!isStructured) {
  1381→          builtFields[f.key] = text
  1382→        } else {
  1383→          const annotations = {}
  1384→          for (const ak of annotationKeys) {
  1385→            const v = fieldValues[`${f.key}__${ak}`] || ''
  1386→            if (v) annotations[ak] = v
  1387→          }
  1388→          if (f.field_type === 'example') {
  1389→            const lines = text.split(' ;;; ').map(s => s.trim()).filter(Boolean)
  1390→            const annoByKey = {}
  1391→            Object.entries(annotations).forEach(([k, raw]) => {
  1392→              annoByKey[k] = String(raw).split(' ;;; ').map(s => s.trim())
  1393→            })
  1394→            builtFields[f.key] = lines.map((line, i) => {
  1395→              const ann = {}
  1396→              Object.entries(annoByKey).forEach(([k, arr]) => { if (arr[i]) ann[k] = arr[i] })
  1397→              return { text: line, annotations: ann }
  1398→            })
  1399→          } else {
  1400→            builtFields[f.key] = [{ text, annotations }]
  1401→          }
  1402→        }
  1403→      }
  1404→      await api.createCard({ deck_id: deckId, word: word.trim(), fields: builtFields })
  1405→      setWord(''); setFieldValues({}); setSaved(true); onSaved()
  1406→      setTimeout(() => setSaved(false), 2000)
  1407→    } catch (e) { alert(e.message) } finally { setSaving(false) }
  1408→  }
  1409→
  1410→  return (
  1411→    <div className="card p-5 space-y-3">
  1412→      <div>
  1413→        <label className="section-title block mb-1.5">Word / Vocab *</label>
  1414→        <input className="input" value={word} onChange={e => setWord(e.target.value)}
  1415→          onKeyDown={e => e.key === 'Enter' && handleSave()}
  1416→          placeholder="Enter the target language word" autoComplete="off" />
  1417→      </div>
  1418→      {fields.map(f => {
  1419→        const annotationKeys = getAnnotationKeys(f.phonetics)
  1420→        return (
  1421→          <div key={f.key}>
  1422→            <label className="section-title block mb-1.5">
  1423→              {f.label}
  1424→              {f.field_type === 'example' && (
  1425→                <span className="ml-2 normal-case font-normal" style={{ color: 'var(--text-muted)' }}>
  1426→                  — use {'{{word}}'} to mark cloze
  1427→                </span>
  1428→              )}
  1429→            </label>
  1430→            {f.field_type === 'example' ? (
  1431→              <>
  1432→                <textarea className="input text-sm resize-none" rows={2} value={fieldValues[f.key] || ''}
  1433→                  onChange={e => setFieldValues(v => ({ ...v, [f.key]: e.target.value }))}
  1434→                  placeholder="e.g. She {{loves}} him. ;;; Their {{love}} is eternal." />
  1435→                {annotationKeys.map(ak => (
  1436→                  <div key={ak} className="mt-1.5">
  1437→                    <label className="section-title block mb-1">{ak}</label>
  1438→                    <textarea className="input text-xs resize-none" rows={2}
  1439→                      value={fieldValues[`${f.key}__${ak}`] || ''}
  1440→                      onChange={e => setFieldValues(v => ({ ...v, [`${f.key}__${ak}`]: e.target.value }))}
  1441→                      placeholder={`${ak} — same order, separated by  ;;; `} />
  1442→                  </div>
  1443→                ))}
  1444→              </>
  1445→            ) : (
  1446→              <>
  1447→                <input className="input text-sm" value={fieldValues[f.key] || ''}
  1448→                  onChange={e => setFieldValues(v => ({ ...v, [f.key]: e.target.value }))}
  1449→                  placeholder={f.description || f.label} />
  1450→                {annotationKeys.map(ak => (
  1451→                  <div key={ak} className="mt-1.5">
  1452→                    <label className="section-title block mb-1">{ak}</label>
  1453→                    <input className="input text-xs" value={fieldValues[`${f.key}__${ak}`] || ''}
  1454→                      onChange={e => setFieldValues(v => ({ ...v, [`${f.key}__${ak}`]: e.target.value }))} />
  1455→                  </div>
  1456→                ))}
  1457→              </>
  1458→            )}
  1459→          </div>
  1460→        )
  1461→      })}
  1462→      <button className="btn-primary" disabled={!word.trim() || saving} onClick={handleSave}>
  1463→        {saving ? 'Saving...' : saved ? '✓ Added!' : 'Add Card'}
  1464→      </button>
  1465→    </div>
  1466→  )
  1467→}

  1546→// ── CollisionModal ─────────────────────────────────────────
