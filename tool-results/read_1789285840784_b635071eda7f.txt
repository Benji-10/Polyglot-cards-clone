    58→export function StudyView({ deckId }: { deckId: string }) {
    59→  const { data: deck, isLoading } = useDeck(deckId);
    60→  const { data: tagsData } = useDeckTags(deckId);
    61→  const { setView } = useUi();
    62→  const { settings } = useAppSettings();
    63→  const { toast } = useToast();
    64→
    65→  const [mode, setMode] = useState<"learn" | "freestyle">("learn");
    66→  const [direction, setDirection] = useState<Direction>("targetToSource");
    67→  const [interaction, setInteraction] = useState<
    68→    "passive" | "typing" | "multiple" | "cloze"
    69→  >("passive");
    70→  const [batchSize, setBatchSize] = useState(settings.defaultBatchSize || 20);
    71→  const [randomise, setRandomise] = useState(false);
    72→  const [pool, setPool] = useState<"all" | "seen" | "unseen">("all");
    73→  const [selectedTag, setSelectedTag] = useState<string | null>(null);
    74→  const [phase, setPhase] = useState<Phase>("setup");
    75→
    76→  // Check for a saved session — initialise state directly from localStorage.
    77→  const sessionKey = `polyglot_session_${deckId}`;
    78→  const [savedSession, setSavedSession] = useState(() => {
    79→    if (typeof window === "undefined") return null;
    80→    try {
    81→      const raw = localStorage.getItem(sessionKey);
    82→      if (raw) {
    83→        const parsed = JSON.parse(raw);
    84→        if (parsed.savedAt && Date.now() - parsed.savedAt < 24 * 60 * 60 * 1000) {
    85→          return parsed;
    86→        }
    87→        localStorage.removeItem(sessionKey);
    88→      }
    89→    } catch {
    90→      /* ignore */
    91→    }
    92→    return null;
    93→  });
    94→
    95→  const { data: studyData, isLoading: cardsLoading } = useStudyCards(deckId, {
    96→    mode,
    97→    pool: mode === "freestyle" ? pool : undefined,
    98→    randomise,
    99→    limit: batchSize,
   100→    tag: selectedTag,
   101→  });
   102→
   103→  const [queue, setQueue] = useState<CardData[]>([]);
   104→  const [index, setIndex] = useState(0);
   105→  const [result, setResult] = useState<SessionResult | null>(null);
   106→
   107→  const startSession = () => {
   108→    if (!studyData?.cards?.length) {
   109→      toast({
   110→        title: mode === "learn" ? "Nothing due" : "No cards in pool",
   111→        description:
   112→          mode === "learn"
   113→            ? "You're all caught up! Try Freestyle to review ahead."
   114→            : undefined,
   115→      });
   116→      return;
   117→    }
   118→    setQueue(studyData.cards);
   119→    setIndex(0);
   120→    setResult(null);
   121→    setPhase("studying");
   122→    // Clear any saved session when starting fresh.
   123→    localStorage.removeItem(sessionKey);
   124→    setSavedSession(null);
   125→  };
   126→
   127→  const onComplete = (r: SessionResult) => {
   128→    setResult(r);
   129→    setPhase("complete");
   130→    // Clear saved session on completion.
   131→    localStorage.removeItem(sessionKey);
   132→    setSavedSession(null);
   133→  };
   134→
   135→  const clearSession = () => {
   136→    localStorage.removeItem(sessionKey);
   137→    setSavedSession(null);
   138→    toast({ title: "Previous session cleared." });
   139→  };
   140→
   141→  if (isLoading || !deck) {
   142→    return (
   143→      <div className="p-8 max-w-3xl mx-auto">
   144→        <Skeleton className="h-10 w-48 shimmer mb-4" />
   145→        <Skeleton className="h-64 rounded-2xl shimmer" />
   146→      </div>
   147→    );
   148→  }
   149→
   150→  return (
   151→    <div className="p-4 md:p-6 lg:p-8 max-w-3xl mx-auto">
   152→      <button
   153→        onClick={() => setView({ name: "deck", deckId })}
   154→        className="flex items-center gap-1.5 text-sm text-secondary hover:text-[var(--text-primary)] mb-4 transition-colors"
   155→      >
   156→        <ArrowLeft className="size-4" /> Back to deck
   157→      </button>
   158→
   159→      {phase === "setup" && (
   160→        <StudySetup
   161→          deckName={deck.name}
   162→          targetLanguage={deck.targetLanguage}
   163→          sourceLanguage={deck.sourceLanguage}
   164→          fields={deck.fields}
   165→          mode={mode}
   166→          setMode={setMode}
   167→          direction={direction}
   168→          setDirection={setDirection}
   169→          interaction={interaction}
   170→          setInteraction={setInteraction}
   171→          batchSize={batchSize}
   172→          setBatchSize={setBatchSize}
   173→          randomise={randomise}
   174→          setRandomise={setRandomise}
   175→          pool={pool}
   176→          setPool={setPool}
   177→          tags={tagsData?.tags || []}
   178→          selectedTag={selectedTag}
   179→          setSelectedTag={setSelectedTag}
   180→          dueCount={deck.stats?.due || 0}
   181→          totalCards={studyData?.total || 0}
   182→          loading={cardsLoading}
   183→          onStart={startSession}
   184→          savedSession={savedSession}
   185→          onClearSession={clearSession}
   186→        />
   187→      )}
   188→
   189→      {phase === "studying" && queue.length > 0 && (
   190→        <StudySession
   191→          deckId={deckId}
   192→          cards={queue}
   193→          index={index}
   194→          setIndex={setIndex}
   195→          interaction={interaction}
   196→          direction={direction}
   197→          fields={deck.fields}
   198→          targetLanguage={deck.targetLanguage}
   199→          sourceLanguage={deck.sourceLanguage}
   200→          contextLanguage={deck.contextLanguage}
   201→          strictAccents={deck.strictAccents}
   202→          strictMode={deck.strictMode}
   203→          latinTyping={deck.latinTyping ?? false}
   204→          romanisationField={deck.romanisationField ?? ""}
   205→          onComplete={onComplete}
   206→          onExit={() => setPhase("setup")}
   207→          ttsEnabled={settings.ttsEnabled}
   208→          ttsRate={settings.ttsRate}
   209→          mode={mode}
   210→        />
   211→      )}
   212→
   213→      {phase === "complete" && result && (
   214→        <StudyComplete result={result} onBack={() => setPhase("setup")} />
   215→      )}
   216→    </div>
   217→  );
   218→}

   542→function getAnswer(
   543→  card: CardData,
   544→  direction: Direction,
   545→  fields: BlueprintFieldDef[],
   546→  latinTyping?: boolean,
   547→  romanisationField?: string
   548→): string {
   549→  if (direction === "targetToSource") {
   550→    // Recall the source-language translation.
   551→    const srcField =
   552→      fields.find((f) => f.key === "source_translation") ||
   553→      fields.find((f) => f.key === "definition") ||
   554→      fields.find((f) => f.key === "reading") ||
   555→      fields[0];
   556→    const raw = srcField ? card.fields[srcField.key] : null;
   557→    return raw ? fieldValueToAnnotated(raw)?.text || "" : card.word;
   558→  }
   559→  // sourceToTarget — recall the target-language word.
   560→  if (latinTyping && romanisationField) {
   561→    const romVal = card.fields[romanisationField];
   562→    const romText = fieldValueToAnnotated(romVal)?.text;
   563→    if (romText) return romText;
   564→  }
   565→  return card.word;
   566→}

   568→function StudySession(props: {

   589→  const reviewMut = useReviewCard(props.deckId);
